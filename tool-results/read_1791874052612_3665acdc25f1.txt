    68→async function* withSeriesPlanning(
    69→  source: AsyncGenerator<unknown>,
    70→  runPlan: (answerText: string) => Promise<SeriesPlan | null>,
    71→  signal?: AbortSignal,
    72→): AsyncGenerator<unknown> {
    73→  let answerText = '';
    74→  let ended = false;
    75→  for await (const event of source) {
    76→    yield event;
    77→    if ((event as { type?: string }).type === 'done') ended = true;
    78→    if ((event as { type?: string }).type === 'chunk') {
    79→      const content = (event as { content?: unknown }).content;
    80→      if (typeof content === 'string') answerText += content;
    81→    }
    82→  }
    83→  if (!ended || signal?.aborted) return;
    84→  const seriesPlan = await runPlan(answerText);
    85→  if (seriesPlan) {
    86→    yield {
    87→      type: 'series_plan',
    88→      plan: seriesPlan,
    89→      at: new Date().toISOString(),
    90→    };
    91→  }

   146→/** NDJSON 流式响应（S3b——运行时执行结果流：submission 首行 + chunk / done / stopped / cancelled + state_updated）。 */
   147→function ndjsonResponse(stream: AsyncGenerator<unknown>): HandlerResult {
   148→  const encoder = new TextEncoder();
   149→  const body = new ReadableStream<Uint8Array>({
   150→    async start(controller) {
   151→      try {
   152→        for await (const event of stream) {
   153→          controller.enqueue(encoder.encode(`${JSON.stringify(event)}\n`));
   154→        }
   155→        controller.close();
   156→      } catch (error) {
   157→        controller.error(error instanceof Error ? error : new Error(String(error)));
   158→      }
   159→    },
   160→  });
   161→  return { status: 200, headers: NDJSON_HEADERS, body };
   162→}

   179→async function* withGeneration(
   180→  source: AsyncGenerator<unknown>,
   181→  intent: {
   182→    modality: GenerationModality;
   183→    prompt: string;
   184→    useContext?: boolean;
   185→    /** 指代类（「对应的猫图片」）——提示词优先由上一轮
   186→     * 助手内容解析（getTranscript 回调注入转录）。 */
   187→    referential?: boolean;
   188→    /** 创作参数（画幅 / 视频时长 / 参考图——透传生成网关）。 */
   189→    aspectRatio?: GenerationOptions['aspectRatio'];
   190→    seconds?: GenerationOptions['seconds'];
   191→    referenceImages?: string[];
   192→  },
   193→  signal?: AbortSignal,
   194→  getTopic?: () => string | null,
   195→  getTranscript?: () => Array<{ role: 'user' | 'assistant'; text: string }>,
   196→  experienceId?: string | null,
   197→): AsyncGenerator<unknown> {
   198→  // 回答内容累积（呈现形式类生成的提示词原料——
   199→  // 「用视频展现」这类输入不含画面主题，主题在本轮
   200→  // 回答里）。chunk 事件在流内逐段拼接。
   201→  let answerText = '';
   202→  let ended = false;
   203→  for await (const event of source) {
   204→    yield event;
   205→    if ((event as { type?: string }).type === 'done') ended = true;
   206→    if ((event as { type?: string }).type === 'chunk') {
   207→      const content = (event as { content?: unknown }).content;
   208→      if (typeof content === 'string') answerText += content;
   209→    }
   210→  }
   211→  // 流末（done 之后）触发生成——对话回应先完整呈现。
   212→  if (!ended || signal?.aborted) return;
   213→  // 呈现形式类生成的提示词来源（2026-10-11 修订）：
   214→  // ①视觉模态（图像 / 视频）优先用体验主题（首轮用户
   215→  // 输入，如「猫为什么会发出呼噜声」——短视觉主题，
   216→  // 文生图模型可消费；实测 400 字叙述文被提供方忽略
   217→  // 并回落人物先验）；无主题时回退回答首句。
   218→  // ①′ 指代类（referential——「生成一个对应的猫图片」）：
   219→  // 主体在上一轮助手内容里（「对应的」指代上文）——优先
   220→  // 由转录的上一轮助手首句解析，主题仅作回退。
   221→  // ②语音为朗读语义——直接朗读回答内容，不加前缀。
   222→  let prompt = intent.prompt;
   223→  if (intent.useContext === true) {
   224→    let topic: string | null = null;
   225→    try {
   226→      topic = getTopic?.() ?? null;
   227→    } catch (topicError) {
   228→      console.error('[withGeneration] getTopic failed:', topicError);
   229→      topic = null;
   230→    }
   231→    if (intent.modality === 'voice') {
   232→      prompt = answerText.trim().slice(0, 1_000) || intent.prompt;
   233→    } else if (intent.referential === true) {
   234→      let referent: string | null = null;
   235→      try {
   236→        const transcript = getTranscript?.() ?? [];
   237→        // 指针须落在当前用户轮**之前**的最近助手轮——当前轮的
   238→        // 回答在流末先入转录（captureLastResponse），直接取
   239→        // 「最后一条助手」会拿到本轮而非指代所指的上文。
   240→        let lastUserIdx = -1;
   241→        for (let i = transcript.length - 1; i >= 0; i -= 1) {
   242→          if (transcript[i].role === 'user') {
   243→            lastUserIdx = i;
   244→            break;
   245→          }
   246→        }
   247→        const lastAssistant =
   248→          lastUserIdx > 0
   249→            ? transcript
   250→                .slice(0, lastUserIdx)
   251→                .reverse()
   252→                .find((turn) => turn.role === 'assistant')
   253→            : undefined;
   254→        const first = lastAssistant?.text.split(/(?<=[。！？])/)[0]?.trim() ?? '';
   255→        if (first.length > 0) referent = first.slice(0, 60);
   256→      } catch (transcriptError) {
   257→        console.error('[withGeneration] getTranscript failed:', transcriptError);
   258→        referent = null;
   259→      }
   260→      prompt =
   261→        referent ??
   262→        (topic !== null && topic.trim().length > 0
   263→          ? topic.trim().slice(0, 60)
   264→          : prompt);
   265→    } else if (topic !== null && topic.trim().length > 0) {
   266→      prompt = topic.trim().slice(0, 60);
   267→    } else {
   268→      const firstSentence = answerText.trim().split(/(?<=[。！？])/)[0] ?? '';
   269→      prompt =
   270→        firstSentence.trim().length > 0
   271→          ? firstSentence.trim().slice(0, 60)
   272→          : intent.prompt;
   273→    }
   274→  }
   275→  const at = new Date().toISOString();
   276→    try {
   277→      const config = await getEnabledProviderConfigForModality(intent.modality);
   278→      // 提示词引擎（2026-10-12——视觉生成算法层）：先经 LLM
   279→      // 蒸馏——创作轮设计文案 + 系列上下文 → 一条覆盖主体 /
   280→      // 风格 / 构图 / 色调的专业提示词。逐轮设计文案不同 →
   281→      // 蒸馏结果不同 → 多轮作品自然分化；蒸馏失败（无文本
   282→      // 配置 / 超时 / 输出不合格）回退既有规则（呈现类用
   283→      // 主题 / 首句；直接创作用原话+增稠），绝不阻塞生成。
   284→      let finalPrompt =
   285→        intent.useContext === true ? prompt : enrichPrompt(prompt);
   286→      let distilledPrompt: string | null = null;
   287→      let designTextForFallback: string | undefined;
   288→      if (intent.modality !== 'voice') {
   289→        try {
   290→          const textConfig = await getEnabledProviderConfig();
   291→          if (textConfig === null) {
   292→            console.error(
   293→              '[withGeneration] distill skipped: no enabled text provider config',
   294→            );
   295→          }
   296→          if (textConfig) {
   297→            // 系列上下文取本会话的【最终】蒸馏提示词（用户意图
   298→            // 太薄，做连贯性参考会失真）。
   299→            const previousPrompts =
   300→              experienceId !== undefined && experienceId !== null
   301→                ? distilledPromptsByExperience.get(experienceId)
   302→                : undefined;
   303→            // 蒸馏设计文案（2026-10-13 CR-56）：呈现形式类生成
   304→            // （useContext——「画成图片」等）的 REPEAT 展示层已
   305→            // 收敛为简短引导，流内 answerText 不再是完整内容——
   306→            // 蒸馏原料由 resolveDistillDesignText 独立从转录取
   307→            // 上一轮助手原文（REPEAT 不入转录，原文仍可回溯）；
   308→            // 取不到原文时为 undefined（蒸馏只基于意图/主题），
   309→            // 绝不把收敛后的简短引导回喂；指代类与直接创作保持
   310→            // 原逻辑（详见 generation-intent.ts 纯函数与单测）。
   311→            let transcriptForDistill: ReadonlyArray<{
   312→              role: 'user' | 'assistant';
   313→              text: string;
   314→            }> = [];
   315→            try {
   316→              transcriptForDistill = getTranscript?.() ?? [];
   317→            } catch {
   318→              transcriptForDistill = [];
   319→            }
   320→            designTextForFallback = resolveDistillDesignText(
   321→              intent,
   322→              answerText,
   323→              transcriptForDistill,
   324→            );
   325→            distilledPrompt = await distillVisualPrompt(textConfig, {
   326→              modality: intent.modality === 'video' ? 'video' : 'image',
   327→              intentPrompt: prompt,
   328→              designText: designTextForFallback,
   329→              aspectRatio: intent.aspectRatio,
   330→              seconds: intent.seconds,
   331→              hasReference: (intent.referenceImages?.length ?? 0) > 0,
   332→              previousPrompts,
   333→            });
   334→          }
   335→        } catch (distillError) {
   336→          console.error('[withGeneration] distill failed:', (distillError as Error)?.message ?? distillError);
   337→          distilledPrompt = null;
   338→        }
   339→        if (distilledPrompt !== null) {
   340→          finalPrompt = distilledPrompt;
   341→          if (experienceId !== undefined && experienceId !== null) {
   342→            const history = distilledPromptsByExperience.get(experienceId) ?? [];
   343→            history.push(distilledPrompt);
   344→            distilledPromptsByExperience.set(experienceId, history.slice(-5));
   345→          }
   346→          console.log('[withGeneration] distilled prompt:', distilledPrompt);
   347→        } else {
   348→          // 蒸馏失败 / 不合格的安全兜底（CR-59）：机制问句直接作
   349→          // 图像提示会引导模型画解剖示意（猫呼噜实测：颈部发光项圈 /
   350→          // 彩色管道剖面 / 箭头标注 / 乱码文字）——兜底收敛为
   351→          // 「主体自然状态 + 温馨场景」，绝不把机制问句原样下发；
   352→          // 带参考图时切参考图导向（CR-61——参考图承载主体）。
   353→          finalPrompt = safeVisualFallback(
   354→            prompt,
   355→            designTextForFallback,
   356→            (intent.referenceImages?.length ?? 0) > 0,
   357→          );
   358→          console.warn(
   359→            '[withGeneration] distill failed, safe visual fallback:',
   360→            finalPrompt,
   361→          );
   362→        }
   363→      }
   364→      yield {
   365→        type: 'generation',
   366→        status: 'started',
   367→        modality: intent.modality,
   368→        prompt,
   369→        ...(distilledPrompt !== null ? { distilledPrompt } : {}),
   370→        at,
   371→      };
   372→      if (!config) {
   373→        yield {
   374→          type: 'generation',
   375→          status: 'unconfigured',
   376→          modality: intent.modality,
   377→          message: `尚未启用${intent.modality === 'image' ? '图像' : intent.modality === 'video' ? '视频' : '语音'}模型——到模型配置添加并启用后，再说一次即可生成。`,
   378→          at: new Date().toISOString(),
   379→        };
   380→        return;
   381→      }
   382→      // 进度透传：提供方轮询回调推入队列，
   383→      // 本循环每 400ms drain 一次并随流输出
   384→      // progress 事件——视频生成耗时数十秒
   385→      // 至数分钟，呈现层据此展示进度条。
   386→      const progressQueue: number[] = [];
   387→      const artifactPromise = GENERATION_CALLS[intent.modality](
   388→        config,
   389→        finalPrompt,
   390→        (progress) => {
   391→          progressQueue.push(progress);
   392→        },
   393→        // 创作参数（画幅 / 时长 / 参考图——前端创作参数条透传；
   394→        // 缺省 undefined 保持提供方默认行为）。
   395→        // 参考图（CR-61）：续作 / 图生图意图的 reference_images
   396→        // 此前解析了却未透传生成网关（仅画幅 / 时长）——参考图
   397→        // 从未到达 agnes extra_body.image，图生图退化为纯文生
   398→        // 图（续作实测：参考图「橘猫蜷缩毛毯」生成「虎斑猫端坐
   399→        // 沙发」，仅氛围延续）。此处补齐透传。
   400→        {
   401→          aspectRatio: intent.aspectRatio,
   402→          seconds: intent.seconds,
   403→          ...(intent.referenceImages !== undefined &&
   404→          intent.referenceImages.length > 0
   405→            ? { referenceImages: intent.referenceImages }
   406→            : {}),
   407→        },
   408→      );
   409→      let artifact: Awaited<typeof artifactPromise>;
   410→      let generationError: unknown;
   411→      let settled = false;
   412→      while (!settled) {
   413→        while (progressQueue.length > 0) {
   414→          const progress = progressQueue.shift();
   415→          if (progress !== undefined) {
   416→            yield {
   417→              type: 'generation',
   418→              status: 'progress',
   419→              modality: intent.modality,
   420→              progress,
   421→              at: new Date().toISOString(),
   422→            };
   423→          }
   424→        }
   425→        settled = await Promise.race([
   426→          artifactPromise.then(
   427→            () => true,
   428→            (error: unknown) => {
   429→              generationError = error;
   430→              return true;
   431→            },
   432→          ),
   433→          new Promise<boolean>((resolve) => {
   434→            setTimeout(() => resolve(false), 400);
   435→          }),
   436→        ]);
   437→      }
   438→      if (generationError !== undefined) {
   439→        throw generationError;
   440→      }
   441→      artifact = await artifactPromise;
   442→      yield {
   443→        type: 'generation',
   444→        status: 'completed',
   445→        modality: intent.modality,
   446→        ...(artifact.url !== undefined ? { url: artifact.url } : {}),
   447→        ...(artifact.dataUri !== undefined ? { dataUri: artifact.dataUri } : {}),
   448→        // 实际生成用的提示词（蒸馏成功 = 蒸馏词；失败 = 安全兜底词；
   449→        // 直接创作 = 增稠结果）——作品持久化 caption 的数据源，
   450→        // 保证作品墙标题有画面语义（CR-60）。
   451→        ...(typeof finalPrompt === 'string' && finalPrompt.length > 0
   452→          ? { promptUsed: finalPrompt }
   453→          : {}),
   454→        mime: artifact.mime,
   455→        at: new Date().toISOString(),
   456→      };
   457→    } catch (error) {
   458→      // 人性化文案（体验流 / 直接生成共用同一纪律——
   459→      // 见 humanizeGenerationFailure）。
   460→      const message = humanizeGenerationFailure(error, intent.modality);
   461→      console.error('[withGeneration] generation failed:', message);
   462→      yield { type: 'generation', status: 'failed', modality: intent.modality, message, at: new Date().toISOString() };
   463→    }
   464→}

   466→/**
   467→ * 创作参数抽取（专业生产能力——画幅 / 视频时长随请求
   468→ * 透传到生成网关）：白名单校验，非法值静默忽略——
   469→ * 参数是增强项，绝不影响对话主轴。
   470→ */
   471→const ASPECT_RATIO_VALUES = new Set(['16:9', '9:16', '1:1']);
   472→const VIDEO_SECONDS_VALUES = new Set(['4', '6', '8', '12']);
   473→
   474→export function extractCreationParams(body: {
   475→  aspect_ratio?: unknown;
   476→  seconds?: unknown;
   477→  reference_images?: unknown;
   478→}): GenerationOptions {
   479→  const options: GenerationOptions = {};
   480→  if (
   481→    typeof body.aspect_ratio === 'string' &&
   482→    ASPECT_RATIO_VALUES.has(body.aspect_ratio)
   483→  ) {
   484→    options.aspectRatio = body.aspect_ratio as GenerationOptions['aspectRatio'];
   485→  }
   486→  if (typeof body.seconds === 'string' && VIDEO_SECONDS_VALUES.has(body.seconds)) {
   487→    options.seconds = body.seconds as GenerationOptions['seconds'];
   488→  }
   489→  // 参考图（图生图）：dataURI 白名单校验——仅 data:image/、
   490→  // 解码后 ≤1.5MB、至多 3 张；非法项静默丢弃。
   491→  // CR-61：原按 dataURI 字符串长度 ≤1.5M 过滤——base64 膨胀
   492→  // ≈1.33×，1.2MB PNG 的 dataURI ≈1.7M 字符被误杀，作品墙
   493→  // 「续作」带的服务端落盘参考图（1.1–1.5MB 常态）从未到达
   494→  // 生成网关，图生图退化为纯文生。改为按 base64 负载估算解码
   495→  // 字节（len×3/4，含 padding 略高估——判定保守、不误放）。
   496→  if (Array.isArray(body.reference_images)) {
   497→    const refs = body.reference_images
   498→      .filter((item): item is string => {
   499→        if (typeof item !== 'string' || !item.startsWith('data:image/')) {
   500→          return false;
   501→        }
   502→        const metaEnd = item.indexOf(',');
   503→        if (metaEnd < 0) return false;
   504→        const payloadLength = item.length - metaEnd - 1;
   505→        if (payloadLength <= 0) return false;
   506→        return (payloadLength * 3) / 4 <= 1_500_000;
   507→      })
   508→      .slice(0, 3);
   509→    if (refs.length > 0) options.referenceImages = refs;
   510→  }
   511→  return options;
   512→}

   519→    retryable: error.retryable,
   520→    ...(error.details ? { details: error.details } : {}),
   521→  });

   524→/**
   525→ * S2/S3 执行形态（S3b——纯暴露层）：经同一运行时实例执行
   526→ * 真实运行路径。experience_id + state_version 定位既有
   527→ * 体验（跨会话分支操作 / 续轮提交）；缺省时经
   528→ * resolveIntent → startExperience 建立新体验并提交首轮。
   529→ */
   530→async function handleExecutionForm(
   531→  body: {
   532→    request_id?: unknown;
   533→    session_id?: unknown;
   534→    user_input?: unknown;
   535→    experience_id?: unknown;
   536→    state_version?: unknown;
   537→    semantic_action?: unknown;
   538→    aspect_ratio?: unknown;
   539→    seconds?: unknown;
   540→  },
   541→  input: StreamRequestInput,
   542→): Promise<HandlerResult> {
   543→  const runtime = input.runtime ?? getServerRuntime();
   544→  const requestId =
   545→    typeof body.request_id === 'string' && body.request_id.length > 0 ? body.request_id : null;
   546→  const sessionId =
   547→    typeof body.session_id === 'string' && body.session_id.length > 0 ? body.session_id : null;
   548→  const userInput =
   549→    typeof body.user_input === 'string' && body.user_input.length > 0 ? body.user_input : null;
   550→  if (requestId === null || sessionId === null || userInput === null) {
   551→    return jsonResponse(400, {
   552→      code: 'INVALID_REQUEST',
   553→      message: '请求缺了必要字段——请刷新页面再试一次。',
   554→      error: 'INVALID_REQUEST',
   555→      reason: 'request_id, session_id and user_input are required non-empty strings',
   556→    });
   557→  }
   558→  const experienceId =
   559→    typeof body.experience_id === 'string' && body.experience_id.length > 0 ? body.experience_id : null;
   560→  const stateVersion =
   561→    typeof body.state_version === 'number' && Number.isInteger(body.state_version) ? body.state_version : null;
   562→  const declaredAction =
   563→    typeof body.semantic_action === 'string' && body.semantic_action.length > 0 ? body.semantic_action : null;
   564→  // 创作参数（专业生产能力——画幅 / 视频时长，白名单外静默忽略）。
   565→  const creationParams = extractCreationParams(body);
   566→
   567→  // 既有体验提交（含跨会话分支操作——S3a D-2：新会话经
   568→  // experience_id + state_version 对持久分支记录显式操作，
   569→  // 会话绑定放宽仅限分支操作路径——运行时内校验）。
   570→  if (experienceId !== null && stateVersion !== null) {
   571→    const classification = classifyInput(userInput);
   572→    const generation = recognizeGenerationIntent(userInput);
   573→    // 生成意图确定性路由（暴露层——与记忆操作同一
   574→    // 纪律）：语义动作词表未命中时按生成意图路由——
   575→    // 呈现形式类（useContext，如「用视频展现」）路由
   576→    // 为 REPEAT（重放上一轮回答——呈现层重放恰好
   577→    // 提供生成提示词的内容原料）；带创作主题的
   578→    // （如「做一段深海雾气的视频」）路由为 CREATE；
   579→    // 其余自由文本兜底 DIRECT_ANSWER（LLM 会话承接
   580→    // ——不再升级拒绝死胡同）。再生成特例：分类为
   581→    // CHANGE_DIRECTION（「换一张图片…」）但生成意图
   582→    // 命中时路由 CREATE（换介质/换一张是对产物的
   583→    // 再生成，非探索方向变更——CHANGE_DIRECTION 在
   584→    // WAITING/CREATION 阶段非法，曾以英文原文报错）。
   585→    // 不发明新语义动作，policy_v2.3.0 词汇表不变。
   586→    const routedClassification =
   587→      classification.semanticAction === 'CHANGE_DIRECTION' && generation !== null
   588→        ? 'CREATE'
   589→        : classification.semanticAction;
   590→    const semanticAction =
   591→      declaredAction ??
   592→      (routedClassification !== 'UNKNOWN'
   593→        ? routedClassification
   594→        : generation !== null
   595→          ? (generation.useContext === true ? 'REPEAT' : 'CREATE')
   596→          : 'DIRECT_ANSWER');
   597→    const submission: SubmissionResult = await runtime.submitExperienceEvent({
   598→      experienceId,
   599→      sessionId,
   600→      semanticAction,
   601→      rawInput: userInput,
   602→      expectedStateVersion: stateVersion,
   603→      requestId,
   604→      signal: input.signal,
   605→    });
   606→    if (!submission.ok) {
   607→      // 呈现形式类降级引导：REPEAT 需要上一完成轮
   608→      // 内容（新会话首轮没有可重放的内容）——此时
   609→      // 用户想展现的对象还不存在，给出可行动的
   610→      // 引导而非创建空创作会话。
   611→      if (
   612→        semanticAction === 'REPEAT' &&
   613→        !declaredAction &&
   614→        submission.error.code === 'INVALID_STATE_TRANSITION'
   615→      ) {
   616→        return jsonResponse(400, {
   617→          code: 'INVALID_ACTION',
   618→          message: '还没有可展现的内容——先聊一个话题，再说「用视频 / 图片展现」。',
   619→          error: 'INVALID_ACTION',
   620→          reason: 'presentation intent with no previous content round to present',
   621→        });
   622→      }
   623→      return runtimeErrorResponse(submission.error);
   624→    }
   625→    return ndjsonResponse(
   626→      generation
   627→        ? withGeneration(
   628→            submission.stream,
   629→            { ...generation, ...creationParams },
   630→            input.signal,
   631→            () => runtime.getExperienceTopic(experienceId),
   632→            () => runtime.getExperienceTranscript(experienceId),
   633→            experienceId,
   634→          )
   635→        : submission.stream,
   636→    );
   637→  }
   638→
   639→  // 新体验流程：意图解析（分类权威——分支操作词 / 记忆操作 /
   640→  // 长期记忆显式表达均在解析阶段确定性识别）。
   641→  const resolution = await runtime.resolveIntent({
   642→    sessionId,
   643→    rawInput: userInput,
   644→    requestId: `${requestId}#intent`,
   645→  });
   646→  if (!resolution.ok) {
   647→    return runtimeErrorResponse(resolution.error);
   648→  }
   649→
   650→  // 记忆操作（WITHDRAW / CORRECT / 长期记忆显式表达写入
   651→  // ——S3a D-1）：无体验轴动作——NDJSON 单行结果 + done。
   652→  if (resolution.action === 'memory_operation') {
   653→    const encoder = new TextEncoder();
   654→    const result = resolution;
   655→    const streamBody = new ReadableStream<Uint8Array>({
   656→      async start(controller) {
   657→        controller.enqueue(
   658→          encoder.encode(
   659→            `${JSON.stringify({
   660→              type: 'submission',
   661→              accepted: true,
   662→              session_id: sessionId,
   663→              request_id: requestId,
   664→              intent_id: result.intent.intentId,
   665→              memory_operation: result.memoryOperation ?? null,
   666→              at: new Date().toISOString(),
   667→            })}\n`,
   668→          ),
   669→        );
   670→        controller.enqueue(encoder.encode(`${JSON.stringify({ type: 'done', at: new Date().toISOString() })}\n`));
   671→        controller.close();
   672→      },
   673→    });
   674→    return { status: 200, headers: NDJSON_HEADERS, body: streamBody };
   675→  }
   676→
   677→  // 生成意图（确定性规则识别——运行时意图解析
   678→  // 已将分类未命中的生成输入确定性路由为 CREATE
   679→  // 创作体验——与 WHAT_IF 首轮自动 CREATE 同一
   680→  // 先例，不发明语义动作）。
   681→  const generation = recognizeGenerationIntent(userInput);
   682→
   683→  // 体验建立 + 首轮提交（创作会话路由 / WHAT_IF 首轮自动
   684→  // CREATE 分支——与进程内路径完全一致；无法归类的自由
   685→  // 文本已在 resolveIntent 兜底为 DIRECT_ANSWER）。
   686→  const started = await runtime.startExperience({
   687→    sessionId,
   688→    intentId: resolution.intent.intentId,
   689→    requestId: `${requestId}#experience`,
   690→  });
   691→  if (!started.ok) {
   692→    return runtimeErrorResponse(started.error);
   693→  }
   694→  const routedSemanticAction =
   695→    declaredAction ?? resolution.semanticAction;
   696→  const submission: SubmissionResult = await runtime.submitExperienceEvent({
   697→    experienceId: started.experienceId,
   698→    sessionId,
   699→    semanticAction: routedSemanticAction,
   700→    rawInput: userInput,
   701→    expectedStateVersion: started.stateVersion,
   702→    requestId: `${requestId}#event`,
   703→    signal: input.signal,
   704→  });
   705→  if (!submission.ok) {
   706→    return runtimeErrorResponse(submission.error);
   707→  }
   708→  // 系列规划触发：无生成意图但用户显式要求系列化产出
   709→  // （角色设定 / 分集 / 短剧）——流末结构化为可执行计划。
   710→  const planTriggered =
   711→    generation === null && SERIES_PLAN_PATTERN.test(userInput);
   712→  const runPlan = planTriggered
   713→    ? async (answerText: string): Promise<SeriesPlan | null> => {
   714→        const textConfig = await getEnabledProviderConfig().catch(() => null);
   715→        if (!textConfig) return null;
   716→        const plan = await planSeries(textConfig, { userInput, answerText });
   717→        if (plan) {
   718→          console.log(
   719→            `[withSeriesPlanning] series plan: ${plan.title}（${plan.characters.length} 角色 / ${plan.episodes.length} 集）`,
   720→          );
   721→        }
   722→        return plan;
   723→      }
   724→    : null;
   725→  return ndjsonResponse(
   726→    generation
   727→      ? withGeneration(
   728→          submission.stream,
   729→          { ...generation, ...creationParams },
   730→          input.signal,
   731→          () => runtime.getExperienceTopic(started.experienceId),
   732→          () => runtime.getExperienceTranscript(started.experienceId),
   733→          started.experienceId,
   734→        )
   735→      : runPlan
   736→        ? withSeriesPlanning(submission.stream, runPlan, input.signal)
   737→        : submission.stream,
   738→  );
   739→}

   741→export async function handleExperienceStreamRequest(
   742→  input: StreamRequestInput,
   743→): Promise<HandlerResult> {
   744→  if (input.method !== 'POST') {
   745→    return jsonResponse(405, { error: 'METHOD_NOT_ALLOWED', allowed: ['POST'] });
   746→  }
   747→
   748→  let parsed: unknown;
   749→  try {
   750→    parsed = JSON.parse(input.body);
   751→  } catch {
   752→    return jsonResponse(400, { code: 'INVALID_JSON_BODY', message: '请求格式有问题——请刷新页面再试一次。', error: 'INVALID_JSON_BODY' });
   753→  }
   754→
   755→  const body = parsed as {
   756→    semanticAction?: unknown;
   757→    request_id?: unknown;
   758→    session_id?: unknown;
   759→    user_input?: unknown;
   760→    experience_id?: unknown;
   761→    state_version?: unknown;
   762→    semantic_action?: unknown;
   763→    aspect_ratio?: unknown;
   764→    seconds?: unknown;
   765→  };
   766→
   767→  // S2/S3 执行形态判定（S3b）：会话上下文三要素齐备
   768→  // （request_id + session_id + user_input）→ 运行时执行
   769→  // 路径；否则为 S1 fixture 形态（semanticAction 直送）。
   770→  const executionForm =
   771→    typeof body.request_id === 'string' &&
   772→    body.request_id.length > 0 &&
   773→    typeof body.session_id === 'string' &&
   774→    body.session_id.length > 0 &&
   775→    typeof body.user_input === 'string' &&
   776→    body.user_input.length > 0;
   777→  if (executionForm) {
   778→    return handleExecutionForm(body, input);
   779→  }
   780→
   781→  const semanticAction = body.semanticAction;
   782→  if (typeof semanticAction !== 'string' || semanticAction.length === 0) {
   783→    return jsonResponse(400, {
   784→      code: 'MISSING_SEMANTIC_ACTION',
   785→      message: '请求缺了必要字段——请刷新页面再试一次。',
   786→      error: 'MISSING_SEMANTIC_ACTION',
   787→    });
   788→  }
   789→
   790→  const policy = resolvePolicy(semanticAction);
   791→  if (!policy.ok) {
   792→    return jsonResponse(400, {
   793→      code: policy.code,
   794→      message: '这个操作目前还不支持——请从界面上的选项里选择。',
   795→      error: policy.code,
   796→      semanticAction,
   797→      reason: 'action is not in the S1 frozen policy set; undefined semantics are escalated, not implemented',
   798→    });
   799→  }
   800→
   801→  // 策略 STOP（S1 硬边界）立即终止，无内容分块，不需要内容语料。
   802→  // 语义空缺（C3 G-1…G-7）不在此补写：冻结表之外的动作已在上方拒绝。
   803→  const stopFixture: ChunkFixture = {
   804→    fixtureId: 'synthetic/stop/v1',
   805→    semanticAction: policy.semanticAction,
   806→    policyAction: 'STOP',
   807→    chunks: [],
   808→  };
   809→  const fixtureResolution: ChunkResolution =
   810→    policy.policyAction === 'STOP'
   811→      ? { ok: true, fixture: stopFixture }
   812→      : loadChunkFixture(policy.semanticAction);
   813→  if (!fixtureResolution.ok) {
   814→    return jsonResponse(500, { error: fixtureResolution.code, semanticAction });
   815→  }
   816→  const fixture = fixtureResolution.fixture;
   817→
   818→  const encoder = new TextEncoder();
   819→  const audit = input.audit;
   820→  const signal = input.signal;
   821→
   822→  const streamBody = new ReadableStream<Uint8Array>({
   823→    async start(controller) {
   824→      try {
   825→        for await (const event of createExperienceStream({
   826→          semanticAction: policy.semanticAction,
   827→          policyAction: policy.policyAction,
   828→          fixtureId: fixture.fixtureId,
   829→          chunks: fixture.chunks,
   830→          signal,
   831→          chunkDelayMs: 40,
   832→          audit,
   833→        })) {
   834→          controller.enqueue(encoder.encode(`${JSON.stringify(event)}\n`));
   835→        }
   836→        controller.close();
   837→      } catch (error) {
   838→        controller.error(error instanceof Error ? error : new Error(String(error)));
   839→      }
   840→    },
   841→  });
   842→
   843→  return {
   844→    status: 200,
   845→    headers: NDJSON_HEADERS,
   846→    body: streamBody,
   847→  };
   848→}
   849→
