   117→   117→async function callProvider(
   118→   118→  config: EnabledProviderConfig,
   119→   119→  path: string,
   120→   120→  body: Record<string, unknown>,
   121→   121→): Promise<Response> {
   122→   122→  // 瞬态自动重试一次：网络波动 / 限速 / 队列满 / 服务端临时
   123→   123→  // 错误（免费档常态）——等待 2s 后重发，两次仍失败才向上抛。
   124→   124→  // 间歇性失败在产品内消化，用户无感；不把「服务正忙」甩给
   125→   125→  // 用户自己重试。
   126→   126→  for (let attempt = 1; attempt <= 2; attempt += 1) {
   127→   127→    const isLastAttempt = attempt === 2;
   128→   128→    let response: Response;
   129→   129→    try {
   130→   130→      response = await fetch(`${baseUrl(config.endpoint)}${path}`, {
   131→   131→        method: 'POST',
   132→   132→        headers: headers(config.apiKey),
   133→   133→        body: JSON.stringify(body),
   134→   134→        signal: AbortSignal.timeout(120_000),
   135→   135→      });
   136→   136→    } catch {
   137→   137→      // 网络失败（超时 / 连接中断——dev 冷启动、上游波动均属
   138→   138→      // 此类）：非末次则重试一次；仍失败标记可重试，文案指向
   139→   139→      // 「稍后再试」而非「换说法」。
   140→   140→      if (!isLastAttempt) {
   141→   141→        await sleep(2_000);
   142→   142→        continue;
   143→   143→      }
   144→   144→      throw new GenerationError(
   145→   145→        `generation provider network failure (provider=${config.provider}, modality=${config.modality})`,
   146→   146→        config.modality,
   147→   147→        true,
   148→   148→      );
   149→   149→    }
   150→   150→    if (response.ok) return response;
   151→   151→    // 错误体透传：提供方返回人类可读原因
   152→   152→    // （如「视频队列已满，请稍后重试」）——
   153→   153→    // 丢弃它会让用户面对一个无法行动的
   154→   154→    // 状态码。瞬态失败标记 retryable。
   155→   155→    const detail = await readProviderError(response);
   156→   156→    if (!isLastAttempt && detail.retryable) {
   157→   157→      await sleep(2_000);
   158→   158→      continue;
   159→   159→    }
   160→   160→    const reason = detail.message.length > 0 ? `: ${detail.message}` : '';
   161→   161→    throw new GenerationError(
   162→   162→      `generation provider failed (provider=${config.provider}, modality=${config.modality}, status=${response.status}${reason})`,
   163→   163→      config.modality,
   164→   164→      detail.retryable,
   165→   165→      response.status,
   166→   166→    );
   167→   167→  }
   168→   168→  // 不可达（循环两轮必抛）——类型收口。
   169→   169→  throw new GenerationError(
   170→   170→    `generation provider failed (provider=${config.provider}, modality=${config.modality})`,
   171→   171→    config.modality,
   172→   172→  );
   173→   173→}

   175→   175→// 瞬态失败：限速 / 队列满 / 服务端临时
   176→   176→// 错误——等待后重试通常可恢复（免费档
   177→   177→// 常态，非产品缺陷）。
   178→   178→const RETRYABLE_STATUS = new Set([408, 429, 500, 502, 503, 504]);
   179→   179→const RETRYABLE_CODES = new Set([
   180→   180→  'rate_limit_exceeded',
   181→   181→  'video_queue_full',
   182→   182→  'queue_full',
   183→   183→  'server_overloaded',

   193→   193→/** 提供方错误体解析（message / error /
   194→   194→ * code 三形态——不含 key、不含请求体）。 */
   195→   195→async function readProviderError(response: Response): Promise<ProviderErrorDetail> {
   196→   196→  let message = '';
   197→   197→  let code: string | undefined;
   198→   198→  try {
   199→   199→    const payload = (await response.json()) as {
   200→   200→      message?: unknown;
   201→   201→      error?: unknown;
   202→   202→      code?: unknown;
   203→   203→    };
   204→   204→    if (typeof payload.message === 'string' && payload.message.length > 0) {
   205→   205→      message = payload.message;
   206→   206→    } else if (typeof payload.error === 'string' && payload.error.length > 0) {
   207→   207→      message = payload.error;
   208→   208→    } else if (payload.error !== null && typeof payload.error === 'object') {
   209→   209→      const nested = (payload.error as { message?: unknown }).message;
   210→   210→      if (typeof nested === 'string' && nested.length > 0) message = nested;
   211→   211→    }
   212→   212→    if (typeof payload.code === 'string' && payload.code.length > 0) {
   213→   213→      code = payload.code;
   214→   214→    }
   215→   215→  } catch {
   216→   216→    // 非 JSON 错误体——仅用状态码。
   217→   217→  }
   218→   218→  const retryable =
   219→   219→    RETRYABLE_STATUS.has(response.status) ||
   220→   220→    (code !== undefined && RETRYABLE_CODES.has(code));
   221→   221→  return { message, retryable };
   222→   222→}

   224→   224→function firstDataEntry(parsed: unknown): Record<string, unknown> | null {

   228→   228→    return data[0] as Record<string, unknown>;
   229→   229→  }
   230→   230→  return null;

   233→   233→function extractUrl(entry: Record<string, unknown>): string | undefined {
   234→   234→  const direct = entry.url;
   235→   235→  if (typeof direct === 'string' && direct.length > 0) return direct;

   240→   240→    if (typeof nested === 'string' && nested.length > 0) return nested;
   241→   241→  }
   242→   242→  const output = entry.output;
   243→   243→  if (output !== null && typeof output === 'object') {
   244→   244→    const nested = (output as { url?: unknown }).url;
   245→   245→    if (typeof nested === 'string' && nested.length > 0) return nested;
   246→   246→  }
   247→   247→  return undefined;

   314→   314→async function generateVideoSeedance(
   315→   315→  config: EnabledProviderConfig,
   316→   316→  prompt: string,
   317→   317→  onProgress?: (progress: number) => void,
   318→   318→  _options?: GenerationOptions,
   319→   319→): Promise<GenerationArtifact> {
   320→   320→  const response = await callProvider(config, '/contents/generations/tasks', {
   321→   321→    model: config.model,
   322→   322→    content: [{ type: 'text', text: prompt }],
   323→   323→  });
   324→   324→  let parsed: unknown;
   325→   325→  try {
   326→   326→    parsed = await response.json();
   327→   327→  } catch {
   328→   328→    throw new GenerationError(
   329→   329→      `generation provider response malformed (provider=${config.provider}, modality=video: non-JSON body)`,
   330→   330→      'video',
   331→   331→    );
   332→   332→  }
   333→   333→  // 同步形态兜底（部分提供方直接返回产物 url）。
   334→   334→  const entry = firstDataEntry(parsed);
   335→   335→  const syncUrl =
   336→   336→    (entry !== null ? extractUrl(entry) : undefined) ??
   337→   337→    extractUrl(parsed as Record<string, unknown>);
   338→   338→  if (syncUrl !== undefined) {
   339→   339→    return { url: syncUrl, mime: 'video/mp4' };
   340→   340→  }
   341→   341→  const taskId = extractTaskId(parsed);
   342→   342→  if (taskId === undefined) {
   343→   343→    throw new GenerationError(
   344→   344→      `generation provider response malformed (provider=${config.provider}, modality=video: no artifact url or task id)`,
   345→   345→      'video',
   346→   346→    );
   347→   347→  }
   348→   348→  const pollUrl = `${baseUrl(config.endpoint)}/contents/generations/tasks/${encodeURIComponent(taskId)}`;
   349→   349→  const deadline = Date.now() + VIDEO_POLL_TIMEOUT_MS;
   350→   350→  for (;;) {
   351→   351→    await sleep(VIDEO_POLL_INTERVAL_MS);
   352→   352→    let status: Response;
   353→   353→    try {
   354→   354→      status = await fetch(pollUrl, {
   355→   355→        headers: headers(config.apiKey),
   356→   356→        signal: AbortSignal.timeout(30_000),
   357→   357→      });
   358→   358→    } catch {
   359→   359→      throw new GenerationError(
   360→   360→        `generation provider network failure (provider=${config.provider}, modality=video: task poll)`,
   361→   361→        'video',
   362→   362→      );
   363→   363→    }
   364→   364→    if (!status.ok) {
   365→   365→      const detail = await readProviderError(status);
   366→   366→      if (!detail.retryable) {
   367→   367→        const reason = detail.message.length > 0 ? `: ${detail.message}` : '';
   368→   368→        throw new GenerationError(
   369→   369→          `generation provider failed (provider=${config.provider}, modality=video: task poll status=${status.status}${reason})`,
   370→   370→          'video',
   371→   371→        );
   372→   372→      }
   373→   373→      await sleep(POLL_RETRY_WAIT_MS);
   374→   374→      continue;
   375→   375→    }
   376→   376→    let state: unknown;
   377→   377→    try {
   378→   378→      state = await status.json();
   379→   379→    } catch {
   380→   380→      throw new GenerationError(
   381→   381→        `generation provider response malformed (provider=${config.provider}, modality=video: task poll non-JSON body)`,
   382→   382→        'video',
   383→   383→      );
   384→   384→    }
   385→   385→    const poll = parseSeedancePoll(state);
   386→   386→    if (poll.status === 'succeeded') {
   387→   387→      if (poll.url !== undefined) {
   388→   388→        return { url: poll.url, mime: 'video/mp4' };
   389→   389→      }
   390→   390→      throw new GenerationError(
   391→   391→        `generation provider response malformed (provider=${config.provider}, modality=video: task completed without url)`,
   392→   392→        'video',

   817→   817→    if (!status.ok) {
   818→   818→      // 轮询瞬态失败（限速）：任务仍在
   819→   819→      // 队列里——等待后继续轮询，而非
   820→   820→      // 判任务失败。
   821→   821→      const detail = await readProviderError(status);
   822→   822→      if (!detail.retryable) {
   823→   823→        const reason =
   824→   824→          detail.message.length > 0 ? `: ${detail.message}` : '';
   825→   825→        throw new GenerationError(
   826→   826→          `generation provider failed (provider=${config.provider}, modality=video: task poll status=${status.status}${reason})`,
   827→   827→          'video',
   828→   828→        );
   829→   829→      }
   830→   830→      await sleep(POLL_RETRY_WAIT_MS);
   831→   831→      continue;
   832→   832→    }
   833→   833→    let state: unknown;
   834→   834→    try {
   835→   835→      state = await status.json();
   836→   836→    } catch {
   837→   837→      throw new GenerationError(
   838→   838→        `generation provider response malformed (provider=${config.provider}, modality=video: task poll non-JSON body)`,
   839→   839→        'video',
   840→   840→      );
   841→   841→    }
   842→   842→    const record = state as {
   843→   843→      status?: unknown;
   844→   844→      error?: unknown;
   845→   845→      progress?: unknown;
   846→   846→    };
   847→   847→    if (record.status === 'completed') {
   848→   848→      const stateEntry = firstDataEntry(state);
   849→   849→      const url =
   850→   850→        extractUrl(state as Record<string, unknown>) ??
   851→   851→        (stateEntry !== null ? extractUrl(stateEntry) : undefined);
   852→   852→      if (url !== undefined) {
   853→   853→        return { url, mime: 'video/mp4' };
   854→   854→      }
   855→   855→      throw new GenerationError(
   856→   856→        `generation provider response malformed (provider=${config.provider}, modality=video: task completed without url)`,
   857→   857→        'video',
   858→   858→      );
   859→   859→    }
   860→   860→    if (record.status === 'failed') {
   861→   861→      const failed = state as {
   862→   862→        error?: unknown;
   863→   863→        message?: unknown;
   864→   864→        code?: unknown;
   865→   865→      };
   866→   866→      const reason =
   867→   867→        typeof failed.error === 'string' && failed.error.length > 0
   868→   868→          ? failed.error
   869→   869→          : typeof failed.message === 'string' && failed.message.length > 0
   870→   870→            ? failed.message
   871→   871→            : typeof failed.code === 'string' && failed.code.length > 0
   872→   872→              ? failed.code
   873→   873→              : 'unknown';
   874→   874→      throw new GenerationError(
   875→   875→        `generation provider task failed (provider=${config.provider}, modality=video: ${reason})`,
   876→   876→        'video',
   877→   877→      );
   878→   878→    }
   879→   879→    // 进度透传（提供方任务 progress 0–100——
   880→   880→    // 呈现层据此展示进度条，视频生成耗时
   881→   881→    // 数十秒至数分钟，反馈不可省）。
   882→   882→    if (onProgress !== undefined && typeof record.progress === 'number') {
   883→   883→      onProgress(record.progress);
   884→   884→    }
   885→   885→    if (Date.now() >= deadline) {
   886→   886→      throw new GenerationError(
   887→   887→        `generation provider timeout (provider=${config.provider}, modality=video: task not completed within ${VIDEO_POLL_TIMEOUT_MS / 1000}s)`,
   888→   888→        'video',
   889→   889→      );
   890→   890→    }
   891→   891→  }
   892→   892→}

   895→   895→  'image' | 'voice' | 'video',
   896→   896→  (
   897→   897→    config: EnabledProviderConfig,
   898→   898→    prompt: string,
   899→   899→    onProgress?: (progress: number) => void,
   900→   900→    options?: GenerationOptions,
   901→   901→  ) => Promise<GenerationArtifact>
   902→   902→ > = {
   903→   903→  image: generateImage,
   904→   904→  voice: generateVoice,
   905→   905→  video: generateVideo,
   906→   906→};
