    74→    }
    75→    return `${modalityLabel}生成没有成功——可以换个说法再试一次。（${error.message}）`;

   117→async function callProvider(
   118→  config: EnabledProviderConfig,
   119→  path: string,
   120→  body: Record<string, unknown>,
   121→): Promise<Response> {
   122→  // 瞬态自动重试一次：网络波动 / 限速 / 队列满 / 服务端临时
   123→  // 错误（免费档常态）——等待 2s 后重发，两次仍失败才向上抛。
   124→  // 间歇性失败在产品内消化，用户无感；不把「服务正忙」甩给
   125→  // 用户自己重试。
   126→  for (let attempt = 1; attempt <= 2; attempt += 1) {
   127→    const isLastAttempt = attempt === 2;
   128→    let response: Response;
   129→    try {
   130→      response = await fetch(`${baseUrl(config.endpoint)}${path}`, {
   131→        method: 'POST',
   132→        headers: headers(config.apiKey),
   133→        body: JSON.stringify(body),
   134→        signal: AbortSignal.timeout(120_000),
   135→      });
   136→    } catch {
   137→      // 网络失败（超时 / 连接中断——dev 冷启动、上游波动均属
   138→      // 此类）：非末次则重试一次；仍失败标记可重试，文案指向
   139→      // 「稍后再试」而非「换说法」。
   140→      if (!isLastAttempt) {
   141→        await sleep(2_000);
   142→        continue;
   143→      }
   144→      throw new GenerationError(
   145→        `generation provider network failure (provider=${config.provider}, modality=${config.modality})`,
   146→        config.modality,
   147→        true,
   148→      );
   149→    }
   150→    if (response.ok) return response;
   151→    // 错误体透传：提供方返回人类可读原因
   152→    // （如「视频队列已满，请稍后重试」）——
   153→    // 丢弃它会让用户面对一个无法行动的
   154→    // 状态码。瞬态失败标记 retryable。
   155→    const detail = await readProviderError(response);
   156→    if (!isLastAttempt && detail.retryable) {
   157→      await sleep(2_000);
   158→      continue;
   159→    }
   160→    const reason = detail.message.length > 0 ? `: ${detail.message}` : '';
   161→    throw new GenerationError(
   162→      `generation provider failed (provider=${config.provider}, modality=${config.modality}, status=${response.status}${reason})`,
   163→      config.modality,
   164→      detail.retryable,
   165→      response.status,
   166→    );
   167→  }
   168→  // 不可达（循环两轮必抛）——类型收口。
   169→  throw new GenerationError(
   170→    `generation provider failed (provider=${config.provider}, modality=${config.modality})`,
   171→    config.modality,
   172→  );
   173→}

   175→// 瞬态失败：限速 / 队列满 / 服务端临时
   176→// 错误——等待后重试通常可恢复（免费档
   177→// 常态，非产品缺陷）。
   178→const RETRYABLE_STATUS = new Set([408, 429, 500, 502, 503, 504]);
   179→const RETRYABLE_CODES = new Set([
   180→  'rate_limit_exceeded',
   181→  'video_queue_full',
   182→  'queue_full',
   183→  'server_overloaded',
   184→  'internal_error',
   185→  'try_again',

   188→interface ProviderErrorDetail {

   190→  retryable: boolean;

   193→/** 提供方错误体解析（message / error /
   194→ * code 三形态——不含 key、不含请求体）。 */
   195→async function readProviderError(response: Response): Promise<ProviderErrorDetail> {
   196→  let message = '';
   197→  let code: string | undefined;
   198→  try {
   199→    const payload = (await response.json()) as {
   200→      message?: unknown;
   201→      error?: unknown;
   202→      code?: unknown;
   203→    };
   204→    if (typeof payload.message === 'string' && payload.message.length > 0) {
   205→      message = payload.message;
   206→    } else if (typeof payload.error === 'string' && payload.error.length > 0) {
   207→      message = payload.error;
   208→    } else if (payload.error !== null && typeof payload.error === 'object') {
   209→      const nested = (payload.error as { message?: unknown }).message;
   210→      if (typeof nested === 'string' && nested.length > 0) message = nested;
   211→    }
   212→    if (typeof payload.code === 'string' && payload.code.length > 0) {
   213→      code = payload.code;
   214→    }
   215→  } catch {
   216→    // 非 JSON 错误体——仅用状态码。
   217→  }
   218→  const retryable =
   219→    RETRYABLE_STATUS.has(response.status) ||
   220→    (code !== undefined && RETRYABLE_CODES.has(code));
   221→  return { message, retryable };
   222→}

   224→function firstDataEntry(parsed: unknown): Record<string, unknown> | null {
   225→  if (parsed === null || typeof parsed !== 'object') return null;
   226→  const data = (parsed as { data?: unknown }).data;
   227→  if (Array.isArray(data) && data.length > 0 && data[0] !== null && typeof data[0] === 'object') {
   228→    return data[0] as Record<string, unknown>;
   229→  }
   230→  return null;
   231→}

   233→function extractUrl(entry: Record<string, unknown>): string | undefined {
   234→  const direct = entry.url;
   235→  if (typeof direct === 'string' && direct.length > 0) return direct;
   236→  // 官方文档：任务完成响应的产物地址在 metadata.url。
   237→  const metadata = entry.metadata;
   238→  if (metadata !== null && typeof metadata === 'object') {
   239→    const nested = (metadata as { url?: unknown }).url;
   240→    if (typeof nested === 'string' && nested.length > 0) return nested;
   241→  }
   242→  const output = entry.output;
   243→  if (output !== null && typeof output === 'object') {
   244→    const nested = (output as { url?: unknown }).url;
   245→    if (typeof nested === 'string' && nested.length > 0) return nested;
   246→  }
   247→  return undefined;
   248→}

   250→/** 异步任务轮询间隔与总超时（Agnes 视频官方
   251→ * 建议每 1–2 秒查询一次；生成通常需数十秒）。 */
   252→const VIDEO_POLL_INTERVAL_MS = 1_500;
   253→// 免费档常态：任务入队后排队可达数分钟
   254→// （实测约 2 分钟才开始处理），总超时
   255→// 须容纳排队 + 处理——过短会把正在
   256→// 处理的任务误判为失败。
   257→const VIDEO_POLL_TIMEOUT_MS = 420_000;
   258→// 任务创建瞬态失败重试（免费档限速 / 队列满常见——
   259→// 实测「视频队列已满」可持续 10 分钟以上，退避 4 段
   260→// 约 5 分钟，尽量 ride out 短时队列波动；仍满则把
   261→// 重试指引交给用户）。重试等待期进度 0 =「排队中」。
   262→const VIDEO_CREATE_RETRY_DELAYS_MS = [30_000, 60_000, 90_000, 120_000];
   263→// 轮询自身瞬态失败（限速）的等待——
   264→// 任务仍在队列里，不应因此判失败。
   265→const POLL_RETRY_WAIT_MS = 5_000;
   266→
   267→function sleep(ms: number): Promise<void> {
   268→  return new Promise((resolve) => setTimeout(resolve, ms));
   269→}

   305→/**
   306→ * Seedance（火山方舟豆包）视频契约执行器——异步任务：
   307→ * POST {base}/contents/generations/tasks 创建（content[0].text
   308→ * 承载提示词），保存 task id 后轮询
   309→ * GET {base}/contents/generations/tasks/{id} 直至 succeeded，
   310→ * 产物在 content.video_url。免费档常态排队/限速——错误映射与
   311→ * 退避复用 callProvider / readProviderError（瞬态 status 标记
   312→ * retryable），轮询用与 agnes 相同的间隔 / 超时纪律。
   313→ */
   314→async function generateVideoSeedance(
   315→  config: EnabledProviderConfig,
   316→  prompt: string,
   317→  onProgress?: (progress: number) => void,
   318→  _options?: GenerationOptions,
   319→): Promise<GenerationArtifact> {
   320→  const response = await callProvider(config, '/contents/generations/tasks', {
   321→    model: config.model,
   322→    content: [{ type: 'text', text: prompt }],
   323→  });
   324→  let parsed: unknown;
   325→  try {
   326→    parsed = await response.json();
   327→  } catch {
   328→    throw new GenerationError(
   329→      `generation provider response malformed (provider=${config.provider}, modality=video: non-JSON body)`,
   330→      'video',
   331→    );
   332→  }
   333→  // 同步形态兜底（部分提供方直接返回产物 url）。
   334→  const entry = firstDataEntry(parsed);
   335→  const syncUrl =
   336→    (entry !== null ? extractUrl(entry) : undefined) ??
   337→    extractUrl(parsed as Record<string, unknown>);
   338→  if (syncUrl !== undefined) {
   339→    return { url: syncUrl, mime: 'video/mp4' };
   340→  }
   341→  const taskId = extractTaskId(parsed);
   342→  if (taskId === undefined) {
   343→    throw new GenerationError(
   344→      `generation provider response malformed (provider=${config.provider}, modality=video: no artifact url or task id)`,
   345→      'video',
   346→    );
   347→  }
   348→  const pollUrl = `${baseUrl(config.endpoint)}/contents/generations/tasks/${encodeURIComponent(taskId)}`;
   349→  const deadline = Date.now() + VIDEO_POLL_TIMEOUT_MS;
   350→  for (;;) {
   351→    await sleep(VIDEO_POLL_INTERVAL_MS);
   352→    let status: Response;
   353→    try {
   354→      status = await fetch(pollUrl, {
   355→        headers: headers(config.apiKey),
   356→        signal: AbortSignal.timeout(30_000),
   357→      });
   358→    } catch {
   359→      throw new GenerationError(
   360→        `generation provider network failure (provider=${config.provider}, modality=video: task poll)`,
   361→        'video',
   362→      );
   363→    }
   364→    if (!status.ok) {
   365→      const detail = await readProviderError(status);
   366→      if (!detail.retryable) {
   367→        const reason = detail.message.length > 0 ? `: ${detail.message}` : '';
   368→        throw new GenerationError(
   369→          `generation provider failed (provider=${config.provider}, modality=video: task poll status=${status.status}${reason})`,
   370→          'video',
   371→        );
   372→      }
   373→      await sleep(POLL_RETRY_WAIT_MS);
   374→      continue;
   375→    }
   376→    let state: unknown;
   377→    try {
   378→      state = await status.json();
   379→    } catch {
   380→      throw new GenerationError(
   381→        `generation provider response malformed (provider=${config.provider}, modality=video: task poll non-JSON body)`,
   382→        'video',
   383→      );
   384→    }
   385→    const poll = parseSeedancePoll(state);
   386→    if (poll.status === 'succeeded') {
   387→      if (poll.url !== undefined) {
   388→        return { url: poll.url, mime: 'video/mp4' };
   389→      }
   390→      throw new GenerationError(
   391→        `generation provider response malformed (provider=${config.provider}, modality=video: task completed without url)`,
   392→        'video',
   393→      );
   394→    }
   395→    if (poll.status === 'failed') {
   396→      throw new GenerationError(
   397→        `generation provider task failed (provider=${config.provider}, modality=video: task failed)`,
   398→        'video',
   399→      );
   400→    }
   401→    if (onProgress && poll.progress !== undefined) {
   402→      onProgress(poll.progress);
   403→    }
   404→    if (Date.now() >= deadline) {
   405→      throw new GenerationError(
   406→        `generation provider timeout (provider=${config.provider}, modality=video: task not completed within ${VIDEO_POLL_TIMEOUT_MS / 1000}s)`,
   407→        'video',
   408→      );
   409→    }
   410→  }
   411→}

   413→/** Seedance 轮询响应解析（status / content.video_url / progress）。 */
   414→function parseSeedancePoll(parsed: unknown): {
   415→  status: string;
   416→  progress?: number;
   417→  url?: string;
   418→} {
   419→  if (parsed === null || typeof parsed !== 'object') {
   420→    return { status: '' };
   421→  }
   422→  const record = parsed as Record<string, unknown>;
   423→  const status =
   424→    typeof record.status === 'string'
   425→      ? record.status.toLowerCase()
   426→      : typeof record.state === 'string'
   427→        ? record.state.toLowerCase()
   428→        : '';
   429→  let progress: number | undefined;
   430→  if (typeof record.progress === 'number') progress = record.progress;
   431→  const content = record.content;
   432→  let url: string | undefined;
   433→  if (content !== null && typeof content === 'object') {
   434→    const c = content as Record<string, unknown>;
   435→    if (typeof c.video_url === 'string' && c.video_url.length > 0) url = c.video_url;
   436→    else if (typeof c.url === 'string' && c.url.length > 0) url = c.url;
   437→  }
   438→  return { status, progress, url };
   439→}

   441→/** 取异步任务 id（video_id / id / task_id 兜底——兼容各家）。 */
   442→function extractTaskId(parsed: unknown): string | undefined {
   443→  if (parsed === null || typeof parsed !== 'object') return undefined;
   444→  const record = parsed as Record<string, unknown>;
   445→  for (const key of ['video_id', 'id', 'task_id', 'taskId']) {
   446→    const value = record[key];
   447→    if (typeof value === 'string' && value.length > 0) return value;
   448→  }
   449→  return undefined;
   450→}

   467→  const response = await callProvider(config, '/videos/generations', {
   468→    ...buildZhipuVideoCreate(config.model, prompt, options),
   469→  });
   470→  let parsed: unknown;
   471→  try {
   472→    parsed = await response.json();
   473→  } catch {
   474→    throw new GenerationError(
   475→      `generation provider response malformed (provider=${config.provider}, modality=video: non-JSON body)`,
   476→      'video',
   477→    );
   478→  }
   479→  const entry = firstDataEntry(parsed);
   480→  const syncUrl =
   481→    (entry !== null ? extractUrl(entry) : undefined) ??
   482→    extractUrl(parsed as Record<string, unknown>);
   483→  if (syncUrl !== undefined) {
   484→    return { url: syncUrl, mime: 'video/mp4' };
   485→  }
   486→  const videoId = parseZhipuCreateId(parsed);
   487→  if (videoId === undefined) {
   488→    throw new GenerationError(
   489→      `generation provider response malformed (provider=${config.provider}, modality=video: no artifact url or task id)`,
   490→      'video',
   491→    );
   492→  }
   493→  const pollUrl = `${baseUrl(config.endpoint)}/async-result/${encodeURIComponent(videoId)}`;
   494→  const deadline = Date.now() + VIDEO_POLL_TIMEOUT_MS;
   495→  for (;;) {
   496→    await sleep(VIDEO_POLL_INTERVAL_MS);
   497→    let status: Response;
   498→    try {
   499→      status = await fetch(pollUrl, {
   500→        headers: headers(config.apiKey),
   501→        signal: AbortSignal.timeout(30_000),
   502→      });
   503→    } catch {
   504→      throw new GenerationError(
   505→        `generation provider network failure (provider=${config.provider}, modality=video: task poll)`,
   506→        'video',
   507→      );
   508→    }
   509→    if (!status.ok) {
   510→      const detail = await readProviderError(status);
   511→      if (!detail.retryable) {
   512→        const reason = detail.message.length > 0 ? `: ${detail.message}` : '';
   513→        throw new GenerationError(
   514→          `generation provider failed (provider=${config.provider}, modality=video: task poll status=${status.status}${reason})`,
   515→          'video',
   516→        );
   517→      }
   518→      await sleep(POLL_RETRY_WAIT_MS);
   519→      continue;
   520→    }
   521→    let state: unknown;
   522→    try {
   523→      state = await status.json();
   524→    } catch {
   525→      throw new GenerationError(
   526→        `generation provider response malformed (provider=${config.provider}, modality=video: task poll non-JSON body)`,
   527→        'video',
   528→      );
   529→    }
   530→    const poll = parseZhipuPoll(state);
   531→    if (poll.status.toUpperCase() === 'SUCCESS') {
   532→      if (poll.url !== undefined) {
   533→        return { url: poll.url, mime: 'video/mp4' };
   534→      }
   535→      throw new GenerationError(
   536→        `generation provider response malformed (provider=${config.provider}, modality=video: task completed without url)`,
   537→        'video',
   538→      );
   539→    }
   540→    if (poll.status.toUpperCase() === 'FAIL') {
   541→      throw new GenerationError(
   542→        `generation provider task failed (provider=${config.provider}, modality=video: task failed)`,
   543→        'video',
   544→      );
   545→    }
   546→    if (onProgress && poll.progress !== undefined) {
   547→      onProgress(poll.progress);
   548→    }
   549→    if (Date.now() >= deadline) {
   550→      throw new GenerationError(
   551→        `generation provider timeout (provider=${config.provider}, modality=video: task not completed within ${VIDEO_POLL_TIMEOUT_MS / 1000}s)`,
   552→        'video',
   553→      );
   554→    }
   555→  }
   556→}

   558→export async function generateImage(
   559→  config: EnabledProviderConfig,
   560→  prompt: string,
   561→  _onProgress?: (progress: number) => void,
   562→  options?: GenerationOptions,
   563→): Promise<GenerationArtifact> {
   564→  const size =
   565→    options?.aspectRatio !== undefined
   566→      ? IMAGE_SIZES[options.aspectRatio]
   567→      : undefined;
   568→  const refs = options?.referenceImages ?? [];
   569→  // 参考图透传日志（CR-61 实证链——确认 dataURI 参考图确实到达
   570→  // 生成网关；仅打长度与来源，不含敏感内容）。
   571→  if (refs.length > 0) {
   572→    console.log(
   573→      `[image-gen] referenceImages=${refs.length} first=${refs[0]?.slice(0, 32)}… len=${refs[0]?.length ?? 0}`,
   574→    );
   575→  }
   576→  // 三段调用形态：带画幅 + 带参考图 → 仅参考图（画幅被拒）→
   577→  // 裸调用（参考图也被拒——保底出图，文生图契约兜底）。
   578→  const call = (withSize: boolean, withRefs: boolean) =>
   579→    callProvider(config, '/images/generations', {
   580→      model: config.model,
   581→      prompt,
   582→      n: 1,
   583→      ...(withSize && size !== undefined ? { size } : {}),
   584→      ...(withRefs && refs.length > 0
   585→        ? { extra_body: { image: refs } }
   586→        : {}),
   587→    });
   588→  let response: Response;
   589→  try {
   590→    response = await call(size !== undefined, refs.length > 0);
   591→  } catch (error) {
   592→    // 参数兼容降级：显式画幅被提供方拒绝（400 且非瞬态）
   593→    // → 无画幅重试（仍带参考图——图生图意图优先于画幅）；
   594→    // 再被拒 → 裸重试（丢参考图保底出图）。
   595→    if (
   596→      size !== undefined &&
   597→      error instanceof GenerationError &&
   598→      !error.retryable &&
   599→      error.status === 400
   600→    ) {
   601→      try {
   602→        response = await call(false, refs.length > 0);
   603→      } catch (retryError) {
   604→        if (
   605→          refs.length > 0 &&
   606→          retryError instanceof GenerationError &&
   607→          !retryError.retryable &&
   608→          retryError.status === 400
   609→        ) {
   610→          response = await call(false, false);
   611→        } else {
   612→          throw retryError;
   613→        }
   614→      }
   615→    } else {
   616→      throw error;
   617→    }
   618→  }
   619→  let parsed: unknown;
   620→  try {
   621→    parsed = await response.json();
   622→  } catch {
   623→    throw new GenerationError(
   624→      `generation provider response malformed (provider=${config.provider}, modality=image: non-JSON body)`,
   625→      'image',
   626→    );
   627→  }
   628→  const entry = firstDataEntry(parsed);
   629→  if (entry === null) {
   630→    throw new GenerationError(
   631→      `generation provider response malformed (provider=${config.provider}, modality=image: missing data[0])`,
   632→      'image',
   633→    );
   634→  }
   635→  const url = extractUrl(entry);
   636→  if (url !== undefined) {
   637→    return { url, mime: 'image/png' };
   638→  }
   639→  const b64 = entry.b64_json;
   640→  if (typeof b64 === 'string' && b64.length > 0) {
   641→    return { dataUri: `data:image/png;base64,${b64}`, mime: 'image/png' };
   642→  }
   643→  throw new GenerationError(
   644→    `generation provider response malformed (provider=${config.provider}, modality=image: no url or b64_json)`,
   645→    'image',
   646→  );
   647→}

   649→export async function generateVoice(
   650→  config: EnabledProviderConfig,
   651→  prompt: string,
   652→): Promise<GenerationArtifact> {
   653→  const response = await callProvider(config, '/audio/speech', {
   654→    model: config.model,
   655→    input: prompt,
   656→    voice: 'alloy',
   657→  });
   658→  // 音频字节流——内容类型依提供方（audio/mpeg / wav / opus）。
   659→  const contentType = response.headers.get('content-type') ?? 'audio/mpeg';
   660→  const mime = contentType.split(';')[0].trim() || 'audio/mpeg';
   661→  let buffer: ArrayBuffer;
   662→  try {
   663→    buffer = await response.arrayBuffer();
   664→  } catch {
   665→    throw new GenerationError(
   666→      `generation provider response malformed (provider=${config.provider}, modality=voice: unreadable audio body)`,
   667→      'voice',
   668→    );
   669→  }
   670→  if (buffer.byteLength === 0) {
   671→    throw new GenerationError(
   672→      `generation provider response malformed (provider=${config.provider}, modality=voice: empty audio body)`,
   673→      'voice',
   674→    );
   675→  }
   676→  const base64 = Buffer.from(buffer).toString('base64');
   677→  return { dataUri: `data:${mime};base64,${base64}`, mime };
   678→}

   680→export async function generateVideo(
   681→  config: EnabledProviderConfig,
   682→  prompt: string,
   683→  onProgress?: (progress: number) => void,
   684→  options?: GenerationOptions,
   685→): Promise<GenerationArtifact> {
   686→  // 契约路由：Seedance（火山方舟豆包）用其自有异步任务契约；
   687→  // 否则按 Agnes 契约（下方原实现，纯增量零回归）。
   688→  if (isSeedanceContract(config)) {
   689→    return generateVideoSeedance(config, prompt, onProgress, options);
   690→  }
   691→  // 智谱（CogVideoX / Vidu）——OpenAI 兼容 /videos/generations 契约。
   692→  if (videoProviderOf(config) === 'zhipu') {
   693→    return generateVideoZhipu(config, prompt, onProgress, options);
   694→  }
   695→  // 官方文档契约（Agnes Video 2.5 Flash——docs/agnes-video-25-flash，
   696→  // OpenAI Videos 兼容异步任务接口）：POST /videos 创建任务
   697→  // （mode 为生成模式 text/keyframe/reference；size 固定 "720P"；
   698→  // seconds 为字符串 "4"–"12"），保存 video_id 后轮询
   699→  // GET /agnesapi?video_id=&model_name= 直至 status=completed，
   700→  // 产物在响应顶层 url。注意与 agnes-video-v2.0 的文档是两代
   701→  // 契约（V2.0 用 width/height/num_frames、产物在 metadata.url
   702→  // ——对本模型发 height 会被 400「禁止字段」拒绝，实测 2026-10-11）。
   703→  // 503 = 服务忙（队列满）——按瞬态退避重试处理。
   704→  // 创作参数：画幅 / 时长由前端透传（缺省 16:9 / 4s 保持
   705→  // 原行为）；自定义参数被 400 拒绝时降级默认参数重试
   706→  // （契约容差——时长/画幅档位可能随提供方收窄）。
   707→  const hasCustomParams =
   708→    options?.aspectRatio !== undefined || options?.seconds !== undefined;
   709→  let useCustomParams = hasCustomParams;
   710→  // 系列参考模式：有公开 URL 参考图时走 reference 模式——
   711→  // 会话内全部分集图作为角色 / 场景参考（≤5），视频连续剧
   712→  // 逐集保持角色一致；无参考图回退 text 模式。dataURI 不
   713→  // 适合视频参考（官方要求公开可访问 URL）——过滤之。
   714→  const videoRefs = (options?.referenceImages ?? []).filter((ref) =>
   715→    ref.startsWith('http'),
   716→  );
   717→  let useRefs = videoRefs.length > 0;
   718→  const createTask = () =>
   719→    callProvider(config, '/videos', {
   720→      model: config.model,
   721→      prompt,
   722→      ...(useRefs
   723→        ? { mode: 'reference', images: videoRefs.slice(0, 5) }
   724→        : { mode: 'text' }),
   725→      seconds: useCustomParams && options?.seconds !== undefined ? options.seconds : '4',
   726→      size: '720P',
   727→      aspect_ratio:
   728→        useCustomParams && options?.aspectRatio !== undefined
   729→          ? options.aspectRatio
   730→          : '16:9',
   731→    });
   732→  const retryDelays = [...VIDEO_CREATE_RETRY_DELAYS_MS];
   733→  // 瞬态失败退避重试：限速 / 队列满时
   734→  // 等待比立即失败更接近成功。
   735→  let response: Response;
   736→  for (;;) {
   737→    try {
   738→      response = await createTask();
   739→      break;
   740→    } catch (error) {
   741→      // 参数拒绝降级：自定义画幅 / 时长 / 参考模式被 400 拒绝
   742→      // → 先降参考模式（保底文本生成），再降参数；逐级保出图。
   743→      if (
   744→        error instanceof GenerationError &&
   745→        !error.retryable &&
   746→        error.status === 400
   747→      ) {
   748→        if (useRefs) {
   749→          useRefs = false;
   750→          continue;
   751→        }
   752→        if (useCustomParams) {
   753→          useCustomParams = false;
   754→          continue;
   755→        }
   756→        throw error;
   757→      }
   758→      const retryable =
   759→        error instanceof GenerationError && error.retryable;
   760→      if (!retryable || retryDelays.length === 0) {
   761→        throw error;
   762→      }
   763→      const wait = retryDelays.shift();
   764→      if (wait === undefined) throw error;
   765→      // 重试等待期间保持前端反馈——
   766→      // 进度 0 即「排队中」态，否则
   767→      // 用户面对两分钟无反馈的「生成中」。
   768→      onProgress?.(0);
   769→      await sleep(wait);
   770→    }
   771→  }
   772→  let parsed: unknown;
   773→  try {
   774→    parsed = await response.json();
   775→  } catch {
   776→    throw new GenerationError(
   777→      `generation provider response malformed (provider=${config.provider}, modality=video: non-JSON body)`,
   778→      'video',
   779→    );
   780→  }
   781→  // 同步形态兜底（部分提供方直接返回产物 URL，
   782→  // 如本地 mock——无需轮询）。
   783→  const entry = firstDataEntry(parsed);
   784→  const syncUrl =
   785→    (entry !== null ? extractUrl(entry) : undefined) ??
   786→    extractUrl(parsed as Record<string, unknown>);
   787→  if (syncUrl !== undefined) {
   788→    return { url: syncUrl, mime: 'video/mp4' };
   789→  }
   790→  const videoId = extractVideoId(parsed);
   791→  if (videoId === undefined) {
   792→    throw new GenerationError(
   793→      `generation provider response malformed (provider=${config.provider}, modality=video: no artifact url or task id)`,
   794→      'video',
   795→    );
   796→  }
   797→  // 异步任务轮询（官方建议每 1–2 秒查询一次，
   798→  // 以 status 与 progress 为准——internal_status
   799→  // 为内部字段，不作为完成判据）。
   800→  const origin = new URL(config.endpoint).origin;
   801→  const pollUrl = `${origin}/agnesapi?video_id=${encodeURIComponent(videoId)}&model_name=${encodeURIComponent(config.model)}`;
   802→  const deadline = Date.now() + VIDEO_POLL_TIMEOUT_MS;
   803→  for (;;) {
   804→    await sleep(VIDEO_POLL_INTERVAL_MS);
   805→    let status: Response;
   806→    try {
   807→      status = await fetch(pollUrl, {
   808→        headers: headers(config.apiKey),
   809→        signal: AbortSignal.timeout(30_000),
   810→      });
   811→    } catch {
   812→      throw new GenerationError(
   813→        `generation provider network failure (provider=${config.provider}, modality=video: task poll)`,
   814→        'video',
   815→      );
   816→    }
   817→    if (!status.ok) {
   818→      // 轮询瞬态失败（限速）：任务仍在
   819→      // 队列里——等待后继续轮询，而非
   820→      // 判任务失败。
   821→      const detail = await readProviderError(status);
   822→      if (!detail.retryable) {
   823→        const reason =
   824→          detail.message.length > 0 ? `: ${detail.message}` : '';
   825→        throw new GenerationError(
   826→          `generation provider failed (provider=${config.provider}, modality=video: task poll status=${status.status}${reason})`,
   827→          'video',
   828→        );
   829→      }
   830→      await sleep(POLL_RETRY_WAIT_MS);
   831→      continue;
   832→    }
   833→    let state: unknown;
   834→    try {
   835→      state = await status.json();
   836→    } catch {
   837→      throw new GenerationError(
   838→        `generation provider response malformed (provider=${config.provider}, modality=video: task poll non-JSON body)`,
   839→        'video',
   840→      );
   841→    }
   842→    const record = state as {
   843→      status?: unknown;
   844→      error?: unknown;
   845→      progress?: unknown;
   846→    };
   847→    if (record.status === 'completed') {
   848→      const stateEntry = firstDataEntry(state);
   849→      const url =
   850→        extractUrl(state as Record<string, unknown>) ??
   851→        (stateEntry !== null ? extractUrl(stateEntry) : undefined);
   852→      if (url !== undefined) {
   853→        return { url, mime: 'video/mp4' };
   854→      }
   855→      throw new GenerationError(
   856→        `generation provider response malformed (provider=${config.provider}, modality=video: task completed without url)`,
   857→        'video',
   858→      );
   859→    }
   860→    if (record.status === 'failed') {
   861→      const failed = state as {
   862→        error?: unknown;
   863→        message?: unknown;
   864→        code?: unknown;
   865→      };
   866→      const reason =
   867→        typeof failed.error === 'string' && failed.error.length > 0
   868→          ? failed.error
   869→          : typeof failed.message === 'string' && failed.message.length > 0
   870→            ? failed.message
   871→            : typeof failed.code === 'string' && failed.code.length > 0
   872→              ? failed.code
   873→              : 'unknown';
   874→      throw new GenerationError(
   875→        `generation provider task failed (provider=${config.provider}, modality=video: ${reason})`,
   876→        'video',
   877→      );
   878→    }
   879→    // 进度透传（提供方任务 progress 0–100——
   880→    // 呈现层据此展示进度条，视频生成耗时
   881→    // 数十秒至数分钟，反馈不可省）。
   882→    if (onProgress !== undefined && typeof record.progress === 'number') {
   883→      onProgress(record.progress);
   884→    }
   885→    if (Date.now() >= deadline) {
   886→      throw new GenerationError(
   887→        `generation provider timeout (provider=${config.provider}, modality=video: task not completed within ${VIDEO_POLL_TIMEOUT_MS / 1000}s)`,
   888→        'video',
   889→      );
   890→    }
   891→  }
   892→}
